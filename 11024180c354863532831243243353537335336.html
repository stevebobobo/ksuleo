📌 範例一：列舉 (enum) 與狀態機 Switch-Case 控制

enum_state.c

示範利用 C 語言的 enum 定義狀態常數 (RUNNING, STOPPED, FAILED, HIBERNATING)，並使用無限迴圈與 switch-case 處理使用者控制輸入。

enum_state.c
/* * 110學年度 C語言範例：列舉 (enum) 與狀態機應用 * 指導/紀錄：Austin Tang */ #include <stdio.h> #include <stdlib.h> // 定義機器運行狀態的 ASCII 列舉值 enum STATE { RUNNING = 49, // '1' STOPPED = 50, // '2' FAILED = 51, // '3' HIBERNATING = 52, // '4' WARM_UP = 53 // '5' }; int main(int argc, char *argv[]) { int input1; printf("RUNNING = %d\n", RUNNING); printf("STOPPED = %d\n", STOPPED); printf("FAILED = %d\n", FAILED); printf("HIBERNATING = %d\n", HIBERNATING); printf("WARM_UP = %d\n", WARM_UP); while(1) { printf("\nPlease provide integer in range [1-5]: "); input1 = getchar(); switch(input1) { case RUNNING: printf("運行中 Machine is RUNNING\n"); break; case STOPPED: printf("停機 Machine is STOPPED\n"); break; case FAILED: printf("失敗 Machine is FAILED\n"); break; case WARM_UP: case HIBERNATING: printf("待機或熱機 Machine is HIBERNATING\n"); break; case 10: // Enter 鍵忽略 break; default: printf("Unknown command!\n"); return 0; } system("pause"); } return 0; }

📌 範例二：結構體 (struct) 陣列與 BMI 指數計算

bmi_struct.c

示範宣告自訂結構體 struct student，建立學生資料陣列，動態輸入身高體重、計算 BMI 指數與全班平均。

bmi_struct.c
/* * 110學年度 C語言範例：計算 BMI (struct 結構體應用) */ #include <stdio.h> #include <stdlib.h> #define NUM 2 // 定義學生人數陣列大小 // 宣告學生結構體 struct student { char name[8]; float Height; // 公尺 int weight; // 公斤 float BMI; }; int main() { struct student C_Class[NUM]; int Member; float total = 0.0f; float Average; printf("Input personal data...\n\n"); for (Member = 0; Member < NUM; Member++) { printf("Number #%d :\n", Member + 1); printf(" Name ? "); scanf("%s", C_Class[Member].name); printf(" Height (m) ? "); scanf("%f", &C_Class[Member].Height); printf(" Weight (kg) ? "); scanf("%d", &C_Class[Member].weight); printf("\n"); } printf("\nDATA LISTING ...\n\n"); printf(" NUMBER NAME HEIGHT WEIGHT BMI\n\n"); for (Member = 0; Member < NUM; Member++) { C_Class[Member].BMI = C_Class[Member].weight / (C_Class[Member].Height * C_Class[Member].Height); printf(" #%d %8s %8.2f %6d %6.2f\n", Member + 1, C_Class[Member].name, C_Class[Member].Height, C_Class[Member].weight, C_Class[Member].BMI); total += C_Class[Member].BMI; } Average = total / NUM; printf("\n Average BMI : %.2f\n\n", Average); system("PAUSE"); return 0; }

📌 範例三：多重迴圈圖形繪製 — 三角形演算法

draw_triangles.c

示範利用韓立瑋同學發想的多重 for 迴圈與函式選單，動態繪製正三角形、倒右直角三角形與左直角三角形。

draw_triangles.c
/* * 110學年度 C語言學生實作：繪製各類圖形 * 發想/學生：韓立瑋 (A100E116) */ #include <stdio.h> #define MAX_LENGTH 100 #define NUM_STRINGS 7 void Inverted_Left(); void Inverted_Right_Triangle(); void Right_Triangle(); void Equilateral(); char menu[NUM_STRINGS][MAX_LENGTH] = { "請問您要哪一種形狀？\n", "1. 倒左直角三角形\n", "2. 倒右直角三角形\n", "3. 右直角三角形\n", "4. 正三角形\n", "0. 離開\n" }; int main() { int choice = 0; while(1) { for (int i = 0; i < NUM_STRINGS - 1; ++i) { printf("%s", menu[i]); } scanf("%d", &choice); if (choice == 0) { printf("已離開！\n"); break; } switch(choice) { case 1: Inverted_Left(); break; case 2: Inverted_Right_Triangle(); break; case 3: Right_Triangle(); break; case 4: Equilateral(); break; default: printf("輸入錯誤！\n"); break; } } return 0; } void Equilateral() { // 正三角形 int cnt1, cnt2; int Max_White = 7; for(cnt1 = Max_White; cnt1 > 0; cnt1--) { for(cnt2 = 0; cnt2 < cnt1; cnt2++) printf(" "); for(cnt2 = (Max_White - cnt1) * 2 - 1; cnt2 > 0; cnt2--) printf("*"); printf("\n"); } }

📌 範例四：個人月儲蓄與收支試算系統

monthly_savings.c

簡沛筑同學作品。定義包含薪資、房租、餐費與水電費之結構體，計算個人每月淨儲蓄與全組平均額度。

monthly_savings.c
/* * 110學年度 C語言學生實作：月儲蓄試算系統 * 作者/學生：簡沛筑 (A100D103) */ #include <stdio.h> #include <stdlib.h> #define NUM 4 struct monthlysavings { char name[8]; int monthlysalary; int rent; int mealcost; int utilitybill; int monthlysavings; }; int main() { struct monthlysavings who[NUM]; int i, total = 0; float Average; printf("Input personal data...\n\n"); for (i = 0; i < NUM; i++) { printf("Number #%d :\n", i + 1); printf(" Who ? "); scanf("%s", who[i].name); printf(" Monthly Salary ? "); scanf("%d", &who[i].monthlysalary); printf(" Rent ? "); scanf("%d", &who[i].rent); printf(" Meal Cost ? "); scanf("%d", &who[i].mealcost); printf(" Utility Bill ? "); scanf("%d", &who[i].utilitybill); } printf("\nDATA LISTING ...\n\n"); for (i = 0; i < NUM; i++) { who[i].monthlysavings = who[i].monthlysalary - (who[i].rent + who[i].mealcost + who[i].utilitybill); printf("#%d %s: Net Savings = %d\n", i + 1, who[i].name, who[i].monthlysavings); total += who[i].monthlysavings; } Average = (float)total / NUM; printf("\nAverage Monthly Savings : %.2f\n", Average); system("PAUSE"); return 0; }

📌 範例五：嵌套雙重迴圈與月曆日期繪製

calendar_loops.c

陳冠良同學作品。運用三層嵌套 for 迴圈走訪年份、月份與日期，並配合條件判斷式輸出日曆格式。

calendar_loops.c
/* * 110學年度 C語言學生實作：嵌套迴圈日曆列印 * 作者/學生：陳冠良 (A100E106) */ #include <stdio.h> int main() { int YearMain = 2021; int UaryMain = 12; int DayMain = 30; for(int Year = 2015; Year <= YearMain; Year++) { printf("\n--------------------------"); printf("\n%d 年", Year); for(int Uary = 1; Uary <= UaryMain; Uary++) { if(Uary == 2) { DayMain = 28; } else if(Uary == 4 || Uary == 6 || Uary == 9 || Uary == 11) { DayMain = 30; } else { DayMain = 31; } printf("\n--------------------------\n %d 月\n--------------------------\n", Uary); for(int Day = 1; Day <= DayMain; Day++) { if(Day % 7 == 1 && Day > 1) { printf("\n"); } printf("%2d ", Day); } printf("\n"); } } return 0; }
