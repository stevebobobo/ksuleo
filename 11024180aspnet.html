📌 專題一：使用 List(Of AccountRecord) 管理帳號與 LINQ 驗證

下載範例 (login_demo.7z)

展示在 ASP.NET Web Forms 中建立自訂 AccountRecord 物件類別，利用 List(Of T) 封裝用戶陣列，並配合 LINQ 與 Regex 正規表達式驗證登入密碼格式。

login_demo.aspx
<%@ Page Language="VB" %> <script runat="server"> ''' <summary> ''' 帳號類別 Account Class ''' </summary> Public Class AccountRecord Public Name As String = "" Public PassWord As String = "" Public Sub init(ByVal name As String, ByVal pss As String) Me.Name = name Me.PassWord = pss End Sub End Class Public AccList As List(Of AccountRecord) Protected Sub InitUserAccount() AccList = New List(Of AccountRecord) Dim BaseAccInfo(,) As String = { {"AAA", "aA1234"}, {"BBB", "bB1234"}, {"CCCC", "bB1234"} } For index = 0 To BaseAccInfo.GetUpperBound(0) Dim user = New AccountRecord() user.init(BaseAccInfo(index, 0), BaseAccInfo(index, 1)) AccList.Add(user) Next End Sub Protected Sub Button1_Click(sender As Object, e As EventArgs) InitUserAccount() ' LINQ 查詢帳號 Dim CurrentUser As AccountRecord = (From UserID In AccList Where UserID.Name = TextBox1.Text).FirstOrDefault() If IsNothing(CurrentUser) Then Label1.Text = TextBox1.Text & "：Can't find Account!" Return End If ' 正規表達式驗證密碼格式 Dim RegexDig6 As Regex = New Regex("^[0-9a-zA-Z_*@\/.+]{6,12}$") If Not RegexDig6.IsMatch(TextBox2.Text) Then Label1.Text = "Wrong password format!" Return End If If TextBox1.Text = CurrentUser.Name And TextBox2.Text = CurrentUser.PassWord Then Label1.Text = "登入成功！" Else Label1.Text = "登入失敗！" End If End Sub </script>

' 110學年度 ASP.NET 茶葉訂購清單系統事件處理 ' 新增項目到 RadioButtonList Protected Sub Button1_Click(sender As Object, e As EventArgs) RadioButtonList1.Items.Add(New ListItem(TextBox1.Text & ", " & TextBox2.Text & "公斤", TextBox2.Text)) TextBox1.Text = "" TextBox2.Text = "" End Sub ' 刪除勾選項目 Protected Sub Button2_Click(sender As Object, e As EventArgs) For Index = RadioButtonList1.Items.Count - 1 To 0 Step -1 If RadioButtonList1.Items(Index).Selected = True Then RadioButtonList1.Items.RemoveAt(Index) End If Next End Sub ' 計算總重量 Protected Sub Button4_Click(sender As Object, e As EventArgs) Dim totalW As Integer = 0 For Index = RadioButtonList1.Items.Count - 1 To 0 Step -1 totalW += Val(RadioButtonList1.Items(Index).Value) Next Label1.Text = "總重：" & totalW & " 公斤" End Sub
